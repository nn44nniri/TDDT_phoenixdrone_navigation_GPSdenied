99bool FGOutputType::Load(Element* element)
100{
101 if (element->FindElementValue("simulation") == string("ON"))
102 SubSystems += ssSimulation;
103 if (element->FindElementValue("aerosurfaces") == string("ON"))
104 SubSystems += ssAerosurfaces;
105 if (element->FindElementValue("rates") == string("ON"))
106 SubSystems += ssRates;
107 if (element->FindElementValue("velocities") == string("ON"))
108 SubSystems += ssVelocities;
109 if (element->FindElementValue("forces") == string("ON"))
110 SubSystems += ssForces;
111 if (element->FindElementValue("moments") == string("ON"))
112 SubSystems += ssMoments;
113 if (element->FindElementValue("atmosphere") == string("ON"))
114 SubSystems += ssAtmosphere;
115 if (element->FindElementValue("massprops") == string("ON"))
116 SubSystems += ssMassProps;
117 if (element->FindElementValue("position") == string("ON"))
118 SubSystems += ssPropagate;
119 if (element->FindElementValue("coefficients") == string("ON") || element->FindElementValue("aerodynamics") == string("ON"))
120 SubSystems += ssAeroFunctions;
121 if (element->FindElementValue("ground_reactions") == string("ON"))
122 SubSystems += ssGroundReactions;
123 if (element->FindElementValue("fcs") == string("ON"))
124 SubSystems += ssFCS;
125 if (element->FindElementValue("propulsion") == string("ON"))
126 SubSystems += ssPropulsion;
127
128 Element *property_element = element->FindElement("property");
129
130 while (property_element) {
131 string property_str = property_element->GetDataLine();
132 SGPropertyNode* node = PropertyManager->GetNode(property_str);
133 if (!node) {
134 FGXMLLogging log(property_element, LogLevel::ERROR);
135 log << LogFormat::RED << LogFormat::BOLD << " No property by the name "
136 << property_str << " has been defined. This property will "
137 << "not be logged. You should check your configuration file.\n"
138 << LogFormat::RESET;
139 } else {
140 if (property_element->HasAttribute("apply")) {
141 string function_str = property_element->GetAttributeValue("apply");
142 auto f = FDMExec->GetTemplateFunc(function_str);
143 if (f)
144 OutputParameters.push_back(new FGFunctionValue(node, f));
145 else {
146 FGXMLLogging log(property_element, LogLevel::ERROR);
147 log << LogFormat::RED << LogFormat::BOLD << " No function by the name "
148 << function_str << " has been defined. This property will "
149 << "not be logged. You should check your configuration file.\n"
150 << LogFormat::RESET;
151 }
152 }
153 else
154 OutputParameters.push_back(new FGPropertyValue(node));
155
156 if (property_element->HasAttribute("caption"))
157 OutputCaptions.push_back(property_element->GetAttributeValue("caption"));
158 else
159 OutputCaptions.push_back("");
160 }
161 property_element = element->FindNextElement("property");
162 }
163
164 double outRate = 1.0;
165 if (element->HasAttribute("rate"))
166 outRate = element->GetAttributeValueAsNumber("rate");
167
168 SetRateHz(outRate);
169
170 return true;
171}

200void FGOutputType::SetRateHz(double rtHz)
201{
202 rtHz = rtHz>1000?1000:(rtHz<0?0:rtHz);
203 if (rtHz > 0) {
204 SetRate(0.5 + 1.0/(FDMExec->GetDeltaT()*rtHz));
205 Enable();
206 } else {
207 SetRate(1);
208 Disable();
209 }
210}

228//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
229// The bitmasked value choices are as follows:
230// unset: In this case (the default) JSBSim would only print
231// out the normally expected messages, essentially echoing
232// the config files as they are read. If the environment
233// variable is not set, debug_lvl is set to 1 internally
234// 0: This requests JSBSim not to output any messages
235// whatsoever.
236// 1: This value explicity requests the normal JSBSim
237// startup messages
238// 2: This value asks for a message to be printed out when
239// a class is instantiated
240// 4: When this value is set, a message is displayed when a
241// FGModel object executes its Run() method
242// 8: When this value is set, various runtime state variables
243// are printed out periodically
244// 16: When set various parameters are sanity checked and
245// a message is printed out when they go out of bounds
246
247void FGOutputType::Debug(int from)
248{
249 if (debug_lvl <= 0) return;
250
251 if (debug_lvl & 1) { // Standard console startup message output
252 if (from == 0) { // Constructor
253 }
254 if (from == 2) {
255 FGLogging log(LogLevel::DEBUG);
256 if (SubSystems & ssSimulation) log << " Simulation parameters logged\n";
257 if (SubSystems & ssAerosurfaces) log << " Aerosurface parameters logged\n";
258 if (SubSystems & ssRates) log << " Rate parameters logged\n";
259 if (SubSystems & ssVelocities) log << " Velocity parameters logged\n";
260 if (SubSystems & ssForces) log << " Force parameters logged\n";
261 if (SubSystems & ssMoments) log << " Moments parameters logged\n";
262 if (SubSystems & ssAtmosphere) log << " Atmosphere parameters logged\n";
263 if (SubSystems & ssMassProps) log << " Mass parameters logged\n";
264 if (SubSystems & ssAeroFunctions) log << " Coefficient parameters logged\n";
265 if (SubSystems & ssPropagate) log << " Propagate parameters logged\n";
266 if (SubSystems & ssGroundReactions) log << " Ground parameters logged\n";
267 if (SubSystems & ssFCS) log << " FCS parameters logged\n";
268 if (SubSystems & ssPropulsion) log << " Propulsion parameters logged\n";
269 if (!OutputParameters.empty()) log << " Properties logged:\n";
270 for (auto param: OutputParameters)
271 log << " - " << param->GetName() << "\n";
272 }
273 }
274 if (debug_lvl & 2 ) { // Instantiation/Destruction notification
275 FGLogging log(LogLevel::DEBUG);
276 if (from == 0) log << "Instantiated: FGOutputType\n";
277 if (from == 1) log << "Destroyed: FGOutputType\n";
278 }
279 if (debug_lvl & 4 ) { // Run() method entry print for FGModel-derived objects
280 }
281 if (debug_lvl & 8 ) { // Runtime state variables
282 }
283 if (debug_lvl & 16) { // Sanity checking
284 }
285 if (debug_lvl & 64) {
286 if (from == 0) { // Constructor
287 }
288 }
289}
