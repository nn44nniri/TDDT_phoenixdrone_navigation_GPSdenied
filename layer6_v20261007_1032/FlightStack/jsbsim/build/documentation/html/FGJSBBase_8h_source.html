123 class Filter {
124 double prev_in;
125 double prev_out;
126 double ca;
127 double cb;
128 public:
129 Filter(void) {}
130 Filter(double coeff, double dt) {
131 prev_in = prev_out = 0.0;
132 double denom = 2.0 + coeff*dt;
133 ca = coeff*dt/denom;
134 cb = (2.0 - coeff*dt)/denom;
135 }
136 double execute(double in) {
137 double out = (in + prev_in)*ca + prev_out*cb;
138 prev_in = in;
139 prev_out = out;
140 return out;
141 }
142 void reset(void) { prev_in = prev_out = 0.0; }
143 };
