78bool FGAtmosphere::InitModel(void)
79{
80 if (!FGModel::InitModel()) return false;
81
82 SLtemperature = Temperature = StdDaySLtemperature;
83 SLpressure = Pressure = StdDaySLpressure;
84 SLdensity = Density = Pressure/(Reng*Temperature);
85 SLsoundspeed = Soundspeed = StdDaySLsoundspeed;
86 Calculate(0.0);
87
88 return true;
89}

107double FGAtmosphere::ValidatePressure(double p, const string& msg, bool quiet) const
108{
109 const double MinPressure = ConvertToPSF(1E-15, ePascals);
110 if (p < MinPressure) {
111 if (!quiet) {
112 FGLogging log(LogLevel::WARN);
113 log << msg << " " << p << " is too low." << endl
114 << msg << " will be capped to " << MinPressure << endl;
115 }
116 return MinPressure;
117 }
118 return p;
119}

121//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
122 // Make sure that the ambient temperature never drops to zero.
123 // According to Wikipedia, 1K is the temperature at the coolest natural place
124 // currently (2023) known in the Universe: the Boomerang Nebula
125
126double FGAtmosphere::ValidateTemperature(double t, const string& msg, bool quiet) const
127{
128 // Minimum known temperature in the universe currently
129 constexpr double minUniverseTemperature = KelvinToRankine(1.0);
130
131 if (t < minUniverseTemperature) {
132 if (!quiet) {
133 FGLogging log(LogLevel::WARN);
134 log << msg << " " << t << " is too low." << endl
135 << msg << " will be capped to " << minUniverseTemperature << endl;
136 }
137 return minUniverseTemperature;
138 }
139 return t;
140}

144void FGAtmosphere::Calculate(double altitude)
145{
146 double t =0.0;
147 double p = 0.0;
148
149 if (!override_node) override_node = atmosphere_node->getNode("override");
150
151 // Temperature and pressure
152 if (override_node) {
153 if (!override_temperature_node)
154 override_temperature_node = override_node->getNode("temperature");
155
156 if (override_temperature_node)
157 t = override_temperature_node->getDoubleValue();
158 else
159 t = GetTemperature(altitude);
160
161 if (!override_pressure_node)
162 override_pressure_node = override_node->getNode("pressure");
163
164 if (override_pressure_node)
165 p = override_pressure_node->getDoubleValue();
166 else
167 p = GetPressure(altitude);
168 } else {
169 t = GetTemperature(altitude);
170 p = GetPressure(altitude);
171 }
172
173 Temperature = ValidateTemperature(t, "", true);
174 Pressure = ValidatePressure(p, "", true);
175
176 // Density
177 if (override_node) {
178 if (!override_density_node)
179 override_density_node = override_node->getNode("density");
180
181 if (override_density_node)
182 Density = override_density_node->getDoubleValue();
183 else
184 Density = Pressure/(Reng*Temperature);
185 }
186 else
187 Density = Pressure/(Reng*Temperature);
188
189 Soundspeed = sqrt(SHRatio*Reng*Temperature);
190 PressureAltitude = CalculatePressureAltitude(Pressure, altitude);
191 DensityAltitude = CalculateDensityAltitude(Density, altitude);
192
193 Viscosity = Beta * pow(Temperature, 1.5) / (SutherlandConstant + Temperature);
194 KinematicViscosity = Viscosity / Density;
195}

238double FGAtmosphere::ConvertToRankine(double t, eTemperature unit) const
239{
240 double targetTemp=0; // in degrees Rankine
241
242 switch(unit) {
243 case eFahrenheit:
244 targetTemp = t + 459.67;
245 break;
246 case eCelsius:
247 targetTemp = (t + 273.15) * 1.8;
248 break;
249 case eRankine:
250 targetTemp = t;
251 break;
252 case eKelvin:
253 targetTemp = t*1.8;
254 break;
255 default:
256 throw BaseException("Undefined temperature unit given");
257 }
258
259 return targetTemp;
260}

290double FGAtmosphere::ConvertToPSF(double p, ePressure unit) const
291{
292 double targetPressure=0; // Pressure in PSF
293
294 switch(unit) {
295 case ePSF:
296 targetPressure = p;
297 break;
298 case eMillibars:
299 targetPressure = p*2.08854342;
300 break;
301 case ePascals:
302 targetPressure = p*0.0208854342;
303 break;
304 case eInchesHg:
305 targetPressure = p*70.7180803;
306 break;
307 default:
308 throw BaseException("Undefined pressure unit given");
309 }
310
311 return targetPressure;

359//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
360// The bitmasked value choices are as follows:
361// unset: In this case (the default) JSBSim would only print
362// out the normally expected messages, essentially echoing
363// the config files as they are read. If the environment
364// variable is not set, debug_lvl is set to 1 internally
365// 0: This requests JSBSim not to output any messages
366// whatsoever.
367// 1: This value explicity requests the normal JSBSim
368// startup messages
369// 2: This value asks for a message to be printed out when
370// a class is instantiated
371// 4: When this value is set, a message is displayed when a
372// FGModel object executes its Run() method
373// 8: When this value is set, various runtime state variables
374// are printed out periodically
375// 16: When set various parameters are sanity checked and
376// a message is printed out when they go out of bounds
377
378void FGAtmosphere::Debug(int from)
379{
380 if (debug_lvl <= 0) return;
381
382 if (debug_lvl & 1) { // Standard console startup message output
383 if (from == 0) { // Constructor
384 }
385 }
386 if (debug_lvl & 2 ) { // Instantiation/Destruction notification
387 FGLogging log(LogLevel::DEBUG);
388 if (from == 0) log << "Instantiated: FGAtmosphere" << std::endl;
389 if (from == 1) log << "Destroyed: FGAtmosphere" << std::endl;
390 }
391 if (debug_lvl & 4 ) { // Run() method entry print for FGModel-derived objects
392 }
393 if (debug_lvl & 8 ) { // Runtime state variables
394 }
395 if (debug_lvl & 16) { // Sanity checking
396 }
397 if (debug_lvl & 128) { //
398 }
399 if (debug_lvl & 64) {
400 if (from == 0) { // Constructor
401 }
402 }
403}
