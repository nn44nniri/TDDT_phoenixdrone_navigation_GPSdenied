141void FGPropagate::SetInitialState(const FGInitialCondition* FGIC)
142{
143 // Initialize the State Vector elements and the transformation matrices
144
145 // Set the position lat/lon/radius
146 VState.vLocation = FGIC->GetPosition();
147
148 epa = FGIC->GetEarthPositionAngleIC();
149 Ti2ec = { cos(epa), sin(epa), 0.0,
150 -sin(epa), cos(epa), 0.0,
151 0.0, 0.0, 1.0 };
152 Tec2i = Ti2ec.Transposed(); // ECEF to ECI frame transform
153
154 VState.vInertialPosition = Tec2i * VState.vLocation;
155
156 UpdateLocationMatrices();
157
158 // Set the orientation from the euler angles (is normalized within the
159 // constructor). The Euler angles represent the orientation of the body
160 // frame relative to the local frame.
161 VState.qAttitudeLocal = FGIC->GetOrientation();
162
163 VState.qAttitudeECI = Ti2l.GetQuaternion()*VState.qAttitudeLocal;
164 UpdateBodyMatrices();
165
166 // Set the velocities in the instantaneus body frame
167 VState.vUVW = FGIC->GetUVWFpsIC();
168
169 // Compute the local frame ECEF velocity
170 vVel = Tb2l * VState.vUVW;
171
172 // Compute local terrain velocity
173 RecomputeLocalTerrainVelocity();
174
175 // Set the angular velocities of the body frame relative to the ECEF frame,
176 // expressed in the body frame.
177 VState.vPQR = FGIC->GetPQRRadpsIC();
178
179 VState.vPQRi = VState.vPQR + Ti2b * in.vOmegaPlanet;
180
181 CalculateInertialVelocity(); // Translational position derivative
182 CalculateQuatdot(); // Angular orientation derivative
183}

196//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
197/*
198Purpose: Called on a schedule to perform EOM integration
199Notes: [JB] Run in standalone mode, SeaLevelRadius will be reference radius.
200 In FGFS, SeaLevelRadius is stuffed from FGJSBSim in JSBSim.cxx each pass.
201
202At the top of this Run() function, see several "shortcuts" (or, aliases) being
203set up for use later, rather than using the longer class->function() notation.
204
205This propagation is done using the current state values
206and current derivatives. Based on these values we compute an approximation to the
207state values for (now + dt).
208
209In the code below, variables named beginning with a small "v" refer to a
210a column vector, variables named beginning with a "T" refer to a transformation
211matrix. ECEF refers to Earth Centered Earth Fixed. ECI refers to Earth Centered
212Inertial.
213
214*/
215
216bool FGPropagate::Run(bool Holding)
217{
218 if (FGModel::Run(Holding)) return true; // Fast return if we have nothing to do ...
219 if (Holding) return false;
220
221 double dt = in.DeltaT * rate; // The 'stepsize'
222
223 // Propagate rotational / translational velocity, angular /translational position, respectively.
224
225 if (!FDMExec->IntegrationSuspended()) {
226 Integrate(VState.qAttitudeECI, VState.vQtrndot, VState.dqQtrndot, dt, integrator_rotational_position);
227 Integrate(VState.vPQRi, in.vPQRidot, VState.dqPQRidot, dt, integrator_rotational_rate);
228 Integrate(VState.vInertialPosition, VState.vInertialVelocity, VState.dqInertialVelocity, dt, integrator_translational_position);
229 Integrate(VState.vInertialVelocity, in.vUVWidot, VState.dqUVWidot, dt, integrator_translational_rate);
230 }
231
232 // CAUTION : the order of the operations below is very important to get
233 // transformation matrices that are consistent with the new state of the
234 // vehicle
235
236 // 1. Update the Earth position angle (EPA)
237 epa += in.vOmegaPlanet(eZ)*dt;
238
239 // 2. Update the Ti2ec and Tec2i transforms from the updated EPA
240 double cos_epa = cos(epa);
241 double sin_epa = sin(epa);
242 Ti2ec = { cos_epa, sin_epa, 0.0,
243 -sin_epa, cos_epa, 0.0,
244 0.0, 0.0, 1.0 };
245 Tec2i = Ti2ec.Transposed(); // ECEF to ECI frame transform
246
247 // 3. Update the location from the updated Ti2ec and inertial position
248 VState.vLocation = Ti2ec*VState.vInertialPosition;
249
250 // 4. Update the other "Location-based" transformation matrices from the
251 // updated vLocation vector.
252 UpdateLocationMatrices();
253
254 // 5. Update the "Orientation-based" transformation matrices from the updated
255 // orientation quaternion and vLocation vector.
256 UpdateBodyMatrices();
257
258 // Translational position derivative (velocities are integrated in the
259 // inertial frame)
260 CalculateUVW();
261
262 // Set auxilliary state variables
263 RecomputeLocalTerrainVelocity();
264
265 VState.vPQR = VState.vPQRi - Ti2b * in.vOmegaPlanet;
266
267 // Angular orientation derivative
268 CalculateQuatdot();
269
270 VState.qAttitudeLocal = Tl2b.GetQuaternion();
271
272 // Compute vehicle velocity wrt ECEF frame, expressed in Local horizontal
273 // frame.
274 vVel = Tb2l * VState.vUVW;
275
276 // Compute orbital parameters in the inertial frame
277 ComputeOrbitalParameters();
278
279 Debug(2);
280 return false;
281}

297//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
298// Compute the quaternion orientation derivative
299//
300// vQtrndot is the quaternion derivative.
301// Reference: See Stevens and Lewis, "Aircraft Control and Simulation",
302// Second edition (2004), eqn 1.5-16b (page 50)
303
304void FGPropagate::CalculateQuatdot(void)
305{
306 // Compute quaternion orientation derivative on current body rates
307 VState.vQtrndot = VState.qAttitudeECI.GetQDot(VState.vPQRi);
308}
309
310//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
311 // Transform the velocity vector of the body relative to the origin (Earth
312 // center) to be expressed in the inertial frame, and add the vehicle velocity
313 // contribution due to the rotation of the planet.
314 // Reference: See Stevens and Lewis, "Aircraft Control and Simulation",
315 // Second edition (2004), eqn 1.5-16c (page 50)
316
317void FGPropagate::CalculateInertialVelocity(void)
318{
319 VState.vInertialVelocity = Tb2i * VState.vUVW + (in.vOmegaPlanet * VState.vInertialPosition);
320}
321
322//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
323 // Transform the velocity vector of the inertial frame to be expressed in the
324 // body frame relative to the origin (Earth center), and substract the vehicle
325 // velocity contribution due to the rotation of the planet.
326
327void FGPropagate::CalculateUVW(void)
328{
329 VState.vUVW = Ti2b * (VState.vInertialVelocity - (in.vOmegaPlanet * VState.vInertialPosition));
330}

334void FGPropagate::Integrate( FGColumnVector3& Integrand,
335 FGColumnVector3& Val,
336 deque <FGColumnVector3>& ValDot,
337 double dt,
338 eIntegrateType integration_type)
339{
340 ValDot.push_front(Val);
341 ValDot.pop_back();
342
343 switch(integration_type) {
344 case eRectEuler: Integrand += dt*ValDot[0];
345 break;
346 case eTrapezoidal: Integrand += 0.5*dt*(ValDot[0] + ValDot[1]);
347 break;
348 case eAdamsBashforth2: Integrand += dt*(1.5*ValDot[0] - 0.5*ValDot[1]);
349 break;
350 case eAdamsBashforth3: Integrand += (1/12.0)*dt*(23.0*ValDot[0] - 16.0*ValDot[1] + 5.0*ValDot[2]);
351 break;
352 case eAdamsBashforth4: Integrand += (1/24.0)*dt*(55.0*ValDot[0] - 59.0*ValDot[1] + 37.0*ValDot[2] - 9.0*ValDot[3]);
353 break;
354 case eAdamsBashforth5: Integrand += dt*((1901./720.)*ValDot[0] - (1387./360.)*ValDot[1] + (109./30.)*ValDot[2] - (637./360.)*ValDot[3] + (251./720.)*ValDot[4]);
355 break;
356 case eNone: // do nothing, freeze translational rate
357 break;
358 case eBuss1:
359 case eBuss2:
360 case eLocalLinearization:
361 {
362 LogException err;
363 err << "Can only use Buss (1 & 2) or local linearization integration methods in for rotational position!";
364 throw err;
365 }
366 default:
367 break;
368 }
369}

372
373void FGPropagate::Integrate( FGQuaternion& Integrand,
374 FGQuaternion& Val,
375 deque <FGQuaternion>& ValDot,
376 double dt,
377 eIntegrateType integration_type)
378{
379 ValDot.push_front(Val);
380 ValDot.pop_back();
381
382 switch(integration_type) {
383 case eRectEuler: Integrand += dt*ValDot[0];
384 break;
385 case eTrapezoidal: Integrand += 0.5*dt*(ValDot[0] + ValDot[1]);
386 break;
387 case eAdamsBashforth2: Integrand += dt*(1.5*ValDot[0] - 0.5*ValDot[1]);
388 break;
389 case eAdamsBashforth3: Integrand += (1/12.0)*dt*(23.0*ValDot[0] - 16.0*ValDot[1] + 5.0*ValDot[2]);
390 break;
391 case eAdamsBashforth4: Integrand += (1/24.0)*dt*(55.0*ValDot[0] - 59.0*ValDot[1] + 37.0*ValDot[2] - 9.0*ValDot[3]);
392 break;
393 case eAdamsBashforth5: Integrand += dt*((1901./720.)*ValDot[0] - (1387./360.)*ValDot[1] + (109./30.)*ValDot[2] - (637./360.)*ValDot[3] + (251./720.)*ValDot[4]);
394 break;
395 case eBuss1:
396 {
397 // This is the first order method as described in Samuel R. Buss paper[6].
398 // The formula from Buss' paper is transposed below to quaternions and is
399 // actually the exact solution of the quaternion differential equation
400 // qdot = 1/2*w*q when w is constant.
401 Integrand = Integrand * QExp(0.5 * dt * VState.vPQRi);
402 }
403 return; // No need to normalize since the quaternion exponential is always normal
404 case eBuss2:
405 {
406 // This is the 'augmented second-order method' from S.R. Buss paper [6].
407 // Unlike Runge-Kutta or Adams-Bashforth, it is a one-pass second-order
408 // method (see reference [6]).
409 FGColumnVector3 wi = VState.vPQRi;
410 FGColumnVector3 wdoti = in.vPQRidot;
411 FGColumnVector3 omega = wi + 0.5*dt*wdoti + dt*dt/12.*wdoti*wi;
412 Integrand = Integrand * QExp(0.5 * dt * omega);
413 }
414 return; // No need to normalize since the quaternion exponential is always normal
415 case eLocalLinearization:
416 {
417 // This is the local linearization algorithm of Barker et al. (see ref. [7])
418 // It is also a one-pass second-order method. The code below is based on the
419 // more compact formulation issued from equation (107) of ref. [8]. The
420 // constants C1, C2, C3 and C4 have the same value than those in ref. [7] pp. 11
421 FGColumnVector3 wi = 0.5 * VState.vPQRi;
422 FGColumnVector3 wdoti = 0.5 * in.vPQRidot;
423 double omegak2 = DotProduct(VState.vPQRi, VState.vPQRi);
424 double omegak = omegak2 > 1E-6 ? sqrt(omegak2) : 1E-6;
425 double rhok = 0.5 * dt * omegak;
426 double C1 = cos(rhok);
427 double C2 = 2.0 * sin(rhok) / omegak;
428 double C3 = 4.0 * (1.0 - C1) / (omegak*omegak);
429 double C4 = 4.0 * (dt - C2) / (omegak*omegak);
430 FGColumnVector3 Omega = C2*wi + C3*wdoti + C4*wi*wdoti;
431 FGQuaternion q;
432
433 q(1) = C1 - C4*DotProduct(wi, wdoti);
434 q(2) = Omega(eP);
435 q(3) = Omega(eQ);
436 q(4) = Omega(eR);
437
438 Integrand = Integrand * q;
439
440 /* Cross check with ref. [7] pp.11-12 formulas and code pp. 20
441 double pk = VState.vPQRi(eP);
442 double qk = VState.vPQRi(eQ);
443 double rk = VState.vPQRi(eR);
444 double pdotk = in.vPQRidot(eP);
445 double qdotk = in.vPQRidot(eQ);
446 double rdotk = in.vPQRidot(eR);
447 double Ap = -0.25 * (pk*pdotk + qk*qdotk + rk*rdotk);
448 double Bp = 0.25 * (pk*qdotk - qk*pdotk);
449 double Cp = 0.25 * (pdotk*rk - pk*rdotk);
450 double Dp = 0.25 * (qk*rdotk - qdotk*rk);
451 double C2p = sin(rhok) / omegak;
452 double C3p = 2.0 * (1.0 - cos(rhok)) / (omegak*omegak);
453 double H = C1 + C4 * Ap;
454 double G = -C2p*rk - C3p*rdotk + C4*Bp;
455 double J = C2p*qk + C3p*qdotk - C4*Cp;
456 double K = C2p*pk + C3p*pdotk - C4*Dp;
457
458 FGLogging log(LogLevel::INFO);
459 log << "q: " << q << "\n";
460
461 // Warning! In the paper of Barker et al. the quaternion components are not
462 // ordered the same way as in JSBSim (see equations (2) and (3) of ref. [7]
463 // as well as the comment just below equation (3))
464 log << "FORTRAN: " << H << " , " << K << " , " << J << " , " << -G << "\n";*/
465 }
466 break; // The quaternion q is not normal so the normalization needs to be done.
467 case eNone: // do nothing, freeze rotational rate
468 break;
469 default:
470 break;
471 }
472
473 Integrand.Normalize();
474}

478void FGPropagate::UpdateLocationMatrices(void)
479{
480 Tl2ec = VState.vLocation.GetTl2ec(); // local to ECEF transform
481 Tec2l = Tl2ec.Transposed(); // ECEF to local frame transform
482 Ti2l = Tec2l * Ti2ec; // ECI to local frame transform
483 Tl2i = Ti2l.Transposed(); // local to ECI transform
484}

488void FGPropagate::UpdateBodyMatrices(void)
489{
490 Ti2b = VState.qAttitudeECI.GetT(); // ECI to body frame transform
491 Tb2i = Ti2b.Transposed(); // body to ECI frame transform
492 Tl2b = Ti2b * Tl2i; // local to body frame transform
493 Tb2l = Tl2b.Transposed(); // body to local frame transform
494 Tec2b = Ti2b * Tec2i; // ECEF to body frame transform
495 Tb2ec = Tec2b.Transposed(); // body to ECEF frame tranform
496
497 Qec2b = Tec2b.GetQuaternion();
498}

502void FGPropagate::ComputeOrbitalParameters(void)
503{
504 const FGColumnVector3 Z{0., 0., 1.};
505 FGColumnVector3 R = VState.vInertialPosition;
506 FGColumnVector3 angularMomentum = R * VState.vInertialVelocity;
507 h = angularMomentum.Magnitude();
508 Inclination = acos(angularMomentum(eZ)/h)*radtodeg;
509 FGColumnVector3 N;
510 if (abs(Inclination) > 1E-8) {
511 N = Z * angularMomentum;
512 RightAscension = atan2(N(eY), N(eX))*radtodeg;
513 N.Normalize();
514 }
515 else {
516 RightAscension = 0.0;
517 N = {1., 0., 0.};
518 PerigeeArgument = 0.0;
519 }
520 R.Normalize();
521 double vr = DotProduct(R, VState.vInertialVelocity);
522 FGColumnVector3 eVector = (VState.vInertialVelocity*angularMomentum/in.GM - R);
523 Eccentricity = eVector.Magnitude();
524 if (Eccentricity > 1E-8) {
525 eVector /= Eccentricity;
526 if (abs(Inclination) > 1E-8) {
527 PerigeeArgument = acos(DotProduct(N, eVector)) * radtodeg;
528 if (eVector(eZ) < 0) PerigeeArgument = 360. - PerigeeArgument;
529 }
530 }
531 else
532 {
533 eVector = {1., 0., 0.};
534 PerigeeArgument = 0.0;
535 }
536
537 TrueAnomaly = acos(Constrain(-1.0, DotProduct(eVector, R), 1.0)) * radtodeg;
538 if (vr < 0.0) TrueAnomaly = 360. - TrueAnomaly;
539 ApoapsisRadius = h*h / (in.GM * (1-Eccentricity));
540 PeriapsisRadius = h*h / (in.GM * (1+Eccentricity));
541
542 if (Eccentricity < 1.0) {
543 double semimajor = 0.5*(ApoapsisRadius + PeriapsisRadius);
544 OrbitalPeriod = 2.*M_PI*pow(semimajor, 1.5)/sqrt(in.GM);
545 }
546 else
547 OrbitalPeriod = 0.0;
548}

742void FGPropagate::WriteStateFile(int num)
743{
744 sg_ofstream outfile;
745
746 if (num == 0) return;
747
748 SGPath path = FDMExec->GetOutputPath();
749
750 if (path.isNull()) path = SGPath("initfile.");
751 else path.append("initfile.");
752
753 // Append sim time to the filename since there may be more than one created during a simulation run
754 path.concat(to_string((double)FDMExec->GetSimTime())+".xml");
755
756 switch(num) {
757 case 1:
758 outfile.open(path);
759 if (outfile.is_open()) {
760 outfile << "<?xml version=\"1.0\"?>\n";
761 outfile << "<initialize name=\"reset00\">\n";
762 outfile << " <ubody unit=\"FT/SEC\"> " << VState.vUVW(eU) << " </ubody> \n";
763 outfile << " <vbody unit=\"FT/SEC\"> " << VState.vUVW(eV) << " </vbody> \n";
764 outfile << " <wbody unit=\"FT/SEC\"> " << VState.vUVW(eW) << " </wbody> \n";
765 outfile << " <phi unit=\"DEG\"> " << VState.qAttitudeLocal.GetEuler(ePhi)*radtodeg << " </phi>\n";
766 outfile << " <theta unit=\"DEG\"> " << VState.qAttitudeLocal.GetEuler(eTht)*radtodeg << " </theta>\n";
767 outfile << " <psi unit=\"DEG\"> " << VState.qAttitudeLocal.GetEuler(ePsi)*radtodeg << " </psi>\n";
768 outfile << " <longitude unit=\"DEG\"> " << VState.vLocation.GetLongitudeDeg() << " </longitude>\n";
769 outfile << " <latitude unit=\"DEG\"> " << VState.vLocation.GetLatitudeDeg() << " </latitude>\n";
770 outfile << " <altitude unit=\"FT\"> " << GetDistanceAGL() << " </altitude>\n";
771 outfile << "</initialize>\n";
772 outfile.close();
773 } else {
774 FGLogging log(LogLevel::ERROR);
775 log << "Could not open and/or write the state to the initial conditions file: "
776 << path << "\n";
777 }
778 break;
779 case 2:
780 outfile.open(path);
781 if (outfile.is_open()) {
782 outfile << "<?xml version=\"1.0\"?>\n";
783 outfile << "<initialize name=\"IC File\" version=\"2.0\">\n";
784 outfile << "\n";
785 outfile << " <position frame=\"ECEF\">\n";
786 outfile << " <latitude unit=\"DEG\" type=\"geodetic\"> " << VState.vLocation.GetGeodLatitudeDeg() << " </latitude>\n";
787 outfile << " <longitude unit=\"DEG\"> " << VState.vLocation.GetLongitudeDeg() << " </longitude>\n";
788 outfile << " <altitudeMSL unit=\"FT\"> " << GetAltitudeASL() << " </altitudeMSL>\n";
789 outfile << " </position>\n";
790 outfile << "\n";
791 outfile << " <orientation unit=\"DEG\" frame=\"LOCAL\">\n";
792 outfile << " <yaw> " << VState.qAttitudeLocal.GetEulerDeg(eYaw) << " </yaw>\n";
793 outfile << " <pitch> " << VState.qAttitudeLocal.GetEulerDeg(ePitch) << " </pitch>\n";
794 outfile << " <roll> " << VState.qAttitudeLocal.GetEulerDeg(eRoll) << " </roll>\n";
795 outfile << " </orientation>\n";
796 outfile << "\n";
797 outfile << " <velocity unit=\"FT/SEC\" frame=\"LOCAL\">\n";
798 outfile << " <x> " << GetVel(eNorth) << " </x>\n";
799 outfile << " <y> " << GetVel(eEast) << " </y>\n";
800 outfile << " <z> " << GetVel(eDown) << " </z>\n";
801 outfile << " </velocity>\n";
802 outfile << "\n";
803 outfile << " <attitude_rate unit=\"DEG/SEC\" frame=\"BODY\">\n";
804 outfile << " <roll> " << (VState.vPQR*radtodeg)(eRoll) << " </roll>\n";
805 outfile << " <pitch> " << (VState.vPQR*radtodeg)(ePitch) << " </pitch>\n";
806 outfile << " <yaw> " << (VState.vPQR*radtodeg)(eYaw) << " </yaw>\n";
807 outfile << " </attitude_rate>\n";
808 outfile << "\n";
809 outfile << "</initialize>\n";
810 outfile.close();
811 } else {
812 FGLogging log(LogLevel::ERROR);
813 log << "Could not open and/or write the state to the initial conditions file: "
814 << path << "\n";
815 }
816 break;
817 default:
818 FGLogging log(LogLevel::ERROR);
819 log << "When writing a state file, the supplied value must be 1 or 2 for the version number of the resulting IC file\n";
820 }
821}
