86// basically P = Q*w and Q_Engine + (-Q_Rotor) = J * dw/dt, J = Moment
87//
88void FGTransmission::Calculate(double EnginePower, double ThrusterTorque, double dt) {
89
90 double coupling = 1.0, coupling_sq = 1.0;
91 double fw_mult = 1.0;
92
93 double d_omega = 0.0, engine_d_omega = 0.0, thruster_d_omega = 0.0; // relative changes
94
95 double engine_omega = rpm_to_omega(EngineRPM);
96 double safe_engine_omega = engine_omega < 1e-1 ? 1e-1 : engine_omega;
97 double engine_torque = EnginePower / safe_engine_omega;
98
99 double thruster_omega = rpm_to_omega(ThrusterRPM);
100 double safe_thruster_omega = thruster_omega < 1e-1 ? 1e-1 : thruster_omega;
101
102 engine_torque -= EngineFriction / safe_engine_omega;
103 ThrusterTorque += Constrain(0.0, BrakeCtrlNorm, 1.0) * MaxBrakePower / safe_thruster_omega;
104
105 // would the FWU release ?
106 engine_d_omega = engine_torque/EngineMoment * dt;
107 thruster_d_omega = - ThrusterTorque/ThrusterMoment * dt;
108
109 if ( thruster_omega+thruster_d_omega > engine_omega+engine_d_omega ) {
110 // don't drive the engine
111 FreeWheelTransmission = 0.0;
112 } else {
113 FreeWheelTransmission = 1.0;
114 }
115
116 fw_mult = FreeWheelLag.execute(FreeWheelTransmission);
117 coupling = fw_mult * Constrain(0.0, ClutchCtrlNorm, 1.0);
118
119 if (coupling < 0.999999) { // are the separate calculations needed ?
120 // assume linear transfer
121 engine_d_omega =
122 (engine_torque - ThrusterTorque*coupling)/(ThrusterMoment*coupling + EngineMoment) * dt;
123 thruster_d_omega =
124 (engine_torque*coupling - ThrusterTorque)/(ThrusterMoment + EngineMoment*coupling) * dt;
125
126 EngineRPM += omega_to_rpm(engine_d_omega);
127 ThrusterRPM += omega_to_rpm(thruster_d_omega);
128
129 // simulate transit to static friction
130 coupling_sq = coupling*coupling;
131 EngineRPM = (1.0-coupling_sq) * EngineRPM + coupling_sq * 0.02 * (49.0*EngineRPM + ThrusterRPM);
132 ThrusterRPM = (1.0-coupling_sq) * ThrusterRPM + coupling_sq * 0.02 * (EngineRPM + 49.0*ThrusterRPM);
133
134 // enforce equal rpm
135 if ( fabs(EngineRPM-ThrusterRPM) < 1e-3 ) {
136 EngineRPM = ThrusterRPM = 0.5 * (EngineRPM+ThrusterRPM);
137 }
138 } else {
139 d_omega = (engine_torque - ThrusterTorque)/(ThrusterMoment + EngineMoment) * dt;
140 EngineRPM = ThrusterRPM += omega_to_rpm(d_omega);
141 }
142
143 // nothing will turn backward
144 if (EngineRPM < 0.0 ) EngineRPM = 0.0;
145 if (ThrusterRPM < 0.0 ) ThrusterRPM = 0.0;
146
147}

151bool FGTransmission::BindModel(int num, FGPropertyManager* PropertyManager)
152{
153 string property_name, base_property_name;
154 base_property_name = CreateIndexedPropertyName("propulsion/engine", num);
155
156 property_name = base_property_name + "/brake-ctrl-norm";
157 PropertyManager->Tie( property_name.c_str(), this,
158 &FGTransmission::GetBrakeCtrlNorm, &FGTransmission::SetBrakeCtrlNorm);
159
160 property_name = base_property_name + "/clutch-ctrl-norm";
161 PropertyManager->Tie( property_name.c_str(), this,
162 &FGTransmission::GetClutchCtrlNorm, &FGTransmission::SetClutchCtrlNorm);
163
164 property_name = base_property_name + "/free-wheel-transmission";
165 PropertyManager->Tie( property_name.c_str(), this,
166 &FGTransmission::GetFreeWheelTransmission);
167
168 return true;
169}

172// The bitmasked value choices are as follows:
173// unset: In this case (the default) JSBSim would only print
174// out the normally expected messages, essentially echoing
175// the config files as they are read. If the environment
176// variable is not set, debug_lvl is set to 1 internally
177// 0: This requests JSBSim not to output any messages
178// whatsoever.
179// 1: This value explicity requests the normal JSBSim
180// startup messages
181// 2: This value asks for a message to be printed out when
182// a class is instantiated
183// 4: When this value is set, a message is displayed when a
184// FGModel object executes its Run() method
185// 8: When this value is set, various runtime state variables
186// are printed out periodically
187// 16: When set various parameters are sanity checked and
188// a message is printed out when they go out of bounds
189
190void FGTransmission::Debug(int from)
191{
192 if (debug_lvl <= 0) return;
193
194 if (debug_lvl & 1) { // Standard console startup message output
195 if (from == 0) { // Constructor
196 }
197 }
198 if (debug_lvl & 2 ) { // Instantiation/Destruction notification
199 FGLogging log(LogLevel::DEBUG);
200 if (from == 0) log << "Instantiated: FGTransmission\n";
201 if (from == 1) log << "Destroyed: FGTransmission\n";
202 }
203 if (debug_lvl & 4 ) { // Run() method entry print for FGModel-derived objects
204 }
205 if (debug_lvl & 8 ) { // Runtime state variables
206 }
207 if (debug_lvl & 16) { // Sanity checking
208 }
209 if (debug_lvl & 64) {
210 if (from == 0) { // Constructor
211 }
212 }
213}
