82FGFDMExec::FGFDMExec(FGPropertyManager* root, std::shared_ptr<unsigned int> fdmctr)
83 : RandomSeed(0), RandomGenerator(make_shared<RandomNumberGenerator>(RandomSeed)),
84 FDMctr(fdmctr)
85{
86 Frame = 0;
87 disperse = 0;
88
89 RootDir = "";
90
91 modelLoaded = false;
92 IsChild = false;
93 holding = false;
94 Terminate = false;
95 HoldDown = false;
96
97 IncrementThenHolding = false; // increment then hold is off by default
98 TimeStepsUntilHold = -1;
99
100 sim_time = 0.0;
101 dT = 1.0/120.0; // a default timestep size. This is needed for when JSBSim is
102 // run in standalone mode with no initialization file.
103
104 AircraftPath = "aircraft";
105 EnginePath = "engine";
106 SystemsPath = "systems";
107
108 if (const char* num = getenv("JSBSIM_DEBUG"); num != nullptr)
109 debug_lvl = strtol(num, nullptr, 0);
110
111 if (!FDMctr) {
112 FDMctr = std::make_shared<unsigned int>(); // Create and initialize the child FDM counter
113 *FDMctr = 0;
114 }
115
116 // Store this FDM's ID
117 IdFDM = *FDMctr; // The main (parent) JSBSim instance is always the "zeroth"
118
119 // Prepare FDMctr for the next child FDM id
120 (*FDMctr)++; // instance. "child" instances are loaded last.
121
122 if (root == nullptr) // Then this is the root FDM
123 Root = new SGPropertyNode();
124 else
125 Root = root->GetNode();
126
127 SGPropertyNode* instanceRoot = Root->getNode("fdm/jsbsim", IdFDM, true);
128 instance = std::make_shared<FGPropertyManager>(instanceRoot);
129
130 if (const char* num = getenv("JSBSIM_DISPERSE");
131 num != nullptr && strtol(num, nullptr, 0) != 0)
132 {
133 disperse = 1; // set dispersions on
134 }
135
136 Debug(0);
137 // this is to catch errors in binding member functions to the property tree.
138 try {
139 Allocate();
140 } catch (const BaseException& e) {
141 LogException err;
142 err << endl << "Caught error: " << e.what() << endl;
143 throw err;
144 }
145
146 trim_status = false;
147 ta_mode = 99;
148 trim_completed = 0;
149
150 Constructing = true;
151 instance->Tie<FGFDMExec, int>("simulation/do_simple_trim", this, nullptr, &FGFDMExec::DoTrim);
152 instance->Tie<FGFDMExec, int>("simulation/do_linearization", this, nullptr, &FGFDMExec::DoLinearization);
153 instance->Tie<FGFDMExec, int>("simulation/reset", this, nullptr, &FGFDMExec::ResetToInitialConditions);
154 instance->Tie("simulation/disperse", this, &FGFDMExec::GetDisperse);
155 instance->Tie("simulation/randomseed", this, &FGFDMExec::SRand, &FGFDMExec::SRand);
156 instance->Tie("simulation/terminate", &Terminate);
157 instance->Tie("simulation/pause", &holding);
158 instance->Tie("simulation/sim-time-sec", this, &FGFDMExec::GetSimTime);
159 instance->Tie("simulation/dt", this, &FGFDMExec::GetDeltaT);
160 instance->Tie("simulation/jsbsim-debug", this, &FGFDMExec::GetDebugLevel, &FGFDMExec::SetDebugLevel);
161 instance->Tie("simulation/frame", reinterpret_cast<int*>(&Frame));
162 instance->Tie("simulation/trim-completed", &trim_completed);
163 instance->Tie("forces/hold-down", this, &FGFDMExec::GetHoldDown, &FGFDMExec::SetHoldDown);
164
165 Constructing = false;
166}

206bool FGFDMExec::Allocate(void)
207{
208 bool result=true;
209
210 Models.resize(eNumStandardModels);
211
212 // First build the inertial model since some other models are relying on
213 // the inertial model and the ground callback to build themselves.
214 // Note that this does not affect the order in which the models will be
215 // executed later.
216 Models[eInertial] = std::make_shared<FGInertial>(this);
217
218 // See the eModels enum specification in the header file. The order of the
219 // enums specifies the order of execution. The Models[] vector is the primary
220 // storage array for the list of models.
221 Models[ePropagate] = std::make_shared<FGPropagate>(this);
222 Models[eInput] = std::make_shared<FGInput>(this);
223 Models[eAtmosphere] = std::make_shared<FGStandardAtmosphere>(this);
224 Models[eWinds] = std::make_shared<FGWinds>(this);
225 Models[eSystems] = std::make_shared<FGFCS>(this);
226 Models[eMassBalance] = std::make_shared<FGMassBalance>(this);
227 Models[eAuxiliary] = std::make_shared<FGAuxiliary>(this);
228 Models[ePropulsion] = std::make_shared<FGPropulsion>(this);
229 Models[eAerodynamics] = std::make_shared<FGAerodynamics> (this);
230 Models[eGroundReactions] = std::make_shared<FGGroundReactions>(this);
231 Models[eExternalReactions] = std::make_shared<FGExternalReactions>(this);
232 Models[eBuoyantForces] = std::make_shared<FGBuoyantForces>(this);
233 Models[eAircraft] = std::make_shared<FGAircraft>(this);
234 Models[eAccelerations] = std::make_shared<FGAccelerations>(this);
235 Models[eOutput] = std::make_shared<FGOutput>(this);
236
237 // Assign the Model shortcuts for internal executive use only.
238 Propagate = static_cast<FGPropagate*>(Models[ePropagate].get());
239 Inertial = static_cast<FGInertial*>(Models[eInertial].get());
240 Input = static_cast<FGInput*>(Models[eInput].get());
241 Atmosphere = static_cast<FGAtmosphere*>(Models[eAtmosphere].get());
242 Winds = static_cast<FGWinds*>(Models[eWinds].get());
243 FCS = static_cast<FGFCS*>(Models[eSystems].get());
244 MassBalance = static_cast<FGMassBalance*>(Models[eMassBalance].get());
245 Auxiliary = static_cast<FGAuxiliary*>(Models[eAuxiliary].get());
246 Propulsion = static_cast<FGPropulsion*>(Models[ePropulsion].get());
247 Aerodynamics = static_cast<FGAerodynamics*>(Models[eAerodynamics].get());
248 GroundReactions = static_cast<FGGroundReactions*>(Models[eGroundReactions].get());
249 ExternalReactions = static_cast<FGExternalReactions*>(Models[eExternalReactions].get());
250 BuoyantForces = static_cast<FGBuoyantForces*>(Models[eBuoyantForces].get());
251 Aircraft = static_cast<FGAircraft*>(Models[eAircraft].get());
252 Accelerations = static_cast<FGAccelerations*>(Models[eAccelerations].get());
253 Output = static_cast<FGOutput*>(Models[eOutput].get());
254
255 // Initialize planet (environment) constants
256 LoadPlanetConstants();
257
258 // Initialize models
259 InitializeModels();
260
261 IC = std::make_shared<FGInitialCondition>(this);
262 IC->bind(instance.get());
263
264 modelLoaded = false;
265
266 return result;
267}

404//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
405
406bool FGFDMExec::Run(void)
407{
408 bool success=true;
409
410 Debug(2);
411
412 for (auto &ChildFDM: ChildFDMList) {
413 ChildFDM->AssignState(Propagate); // Transfer state to the child FDM
414 ChildFDM->Run();
415 }
416
417 IncrTime();
418
419 // returns true if success, false if complete
420 if (Script && !IntegrationSuspended()) success = Script->RunScript();
421
422 for (unsigned int i = 0; i < Models.size(); i++) {
423 LoadInputs(i);
424 Models[i]->Run(holding);
425 }
426
427 if (Terminate) success = false;
428
429 return success;
430}

1033 return false;
1034
1035 element = document->FindNextElement("input");
1036 }
1037

1042 if (!Output->Load(element))
1043 return false;
1044
1045 element = document->FindNextElement("output");
1046 }
1047

1051 result = ReadChild(element);
1052 if (!result) {
1053 FGXMLLogging log(element, LogLevel::ERROR);
1054 log << endl << "Aircraft child element has problems in file " << aircraftCfgFileName << endl;

1056 }
1057 }
1058
1059 // Since all vehicle characteristics have been loaded, place the values in the Inputs
1060 // structure for the FGModel-derived classes.
1061 LoadModelConstants();
1062
1063 modelLoaded = true;
1064
1065 if (IsChild) debug_lvl = saved_debug_lvl;
1066
1067 } else {
1068 FGLogging log(LogLevel::ERROR);
1069 log << LogFormat::RED
1070 << " JSBSim failed to open the configuration file: " << aircraftCfgFileName
1071 << LogFormat::DEFAULT << endl;
1072 }

1074 for (unsigned int i=0; i< Models.size(); i++) LoadInputs(i);
1075
1076 if (result) {
1077 struct PropertyCatalogStructure masterPCS;
1078 masterPCS.base_string = "";
1079 masterPCS.node = Root;
1080 BuildPropertyCatalog(&masterPCS);
1081 }
1082
1083 return result;
1084}

1095void FGFDMExec::BuildPropertyCatalog(struct PropertyCatalogStructure* pcs)
1096{
1097 auto pcsNew = std::make_unique<struct PropertyCatalogStructure>();

1099 const string root_name = GetFullyQualifiedName(root_node) + "/";
1100 const size_t root_name_length = root_name.length();
1101
1102 for (int i=0; i<pcs->node->nChildren(); i++) {
1103 string access="";
1104 pcsNew->base_string = pcs->base_string + "/" + pcs->node->getChild(i)->getNameString();
1105 int node_idx = pcs->node->getChild(i)->getIndex();
1106 if (node_idx != 0) {
1107 pcsNew->base_string = CreateIndexedPropertyName(pcsNew->base_string, node_idx);
1108 }
1109 if (pcs->node->getChild(i)->nChildren() == 0) {
1110 if (pcsNew->base_string.substr(0, root_name_length) == root_name) {
1111 pcsNew->base_string = pcsNew->base_string.erase(0, root_name_length);
1112 }
1113 if (pcs->node->getChild(i)->getAttribute(SGPropertyNode::READ)) access="R";
1114 if (pcs->node->getChild(i)->getAttribute(SGPropertyNode::WRITE)) access+="W";
1115 PropertyCatalog.push_back(pcsNew->base_string+" ("+access+")");
1116 } else {
1117 pcsNew->node = pcs->node->getChild(i);
1118 BuildPropertyCatalog(pcsNew.get());
1119 }
1120 }
