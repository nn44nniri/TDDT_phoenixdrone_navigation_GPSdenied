70FGRotor::FGRotor(FGFDMExec *exec, Element* rotor_element, int num)
71 : FGThruster(exec, rotor_element, num),
72 rho(0.002356), // environment
73 Radius(0.0), BladeNum(0), // configuration parameters
74 Sense(1.0), NominalRPM(0.0), MinimalRPM(0.0), MaximalRPM(0.0),
75 ExternalRPM(0), RPMdefinition(0), ExtRPMsource(NULL), SourceGearRatio(1.0),
76 BladeChord(0.0), LiftCurveSlope(0.0), BladeTwist(0.0), HingeOffset(0.0),
77 BladeFlappingMoment(0.0), BladeMassMoment(0.0), PolarMoment(0.0),
78 InflowLag(0.0), TipLossB(0.0),
79 GroundEffectExp(0.0), GroundEffectShift(0.0), GroundEffectScaleNorm(1.0),
80 LockNumberByRho(0.0), Solidity(0.0), // derived parameters
81 ControlMap(eMainCtrl), // control
82 CollectiveCtrl(0.0), LateralCtrl(0.0), LongitudinalCtrl(0.0),
83 Transmission(NULL), // interaction with engine
84 MaxBrakePower(0.0), GearLoss(0.0), GearMoment(0.0)
85{
86 FGColumnVector3 location(0.0, 0.0, 0.0), orientation(0.0, 0.0, 0.0);
87 Element *thruster_element;
88 double engine_power_est = 0.0;
89
90 // initialise/set remaining variables
91 SetTransformType(FGForce::tCustom);
92 Type = ttRotor;
93 GearRatio = 1.0;
94 InitDynamics();
95
96 dt = exec->GetDeltaT();
97 for (int i=0; i<5; i++) R[i] = 0.0;
98 for (int i=0; i<5; i++) B[i] = 0.0;
99
100 // get positions
101 thruster_element = rotor_element->GetParent()->FindElement("sense");
102 if (thruster_element) {
103 double s = thruster_element->GetDataAsNumber();
104 if (s < -0.1) {
105 Sense = -1.0; // 'CW' as seen from above
106 } else if (s < 0.1) {
107 Sense = 0.0; // 'coaxial'
108 } else {
109 Sense = 1.0; // 'CCW' as seen from above
110 }
111 }
112
113 thruster_element = rotor_element->GetParent()->FindElement("location");
114 if (thruster_element) {
115 location = thruster_element->FindElementTripletConvertTo("IN");
116 } else {
117 FGXMLLogging log(rotor_element, LogLevel::ERROR);
118 log << "No thruster location found.\n";
119 }
120
121 thruster_element = rotor_element->GetParent()->FindElement("orient");
122 if (thruster_element) {
123 orientation = thruster_element->FindElementTripletConvertTo("RAD");
124 } else {
125 FGXMLLogging log(rotor_element, LogLevel::ERROR);
126 log << "No thruster orientation found.\n";
127 }
128
129 SetLocation(location);
130 SetAnglesToBody(orientation);
131 InvTransform = Transform().Transposed(); // body to custom/native
132
133 // wire controls
134 ControlMap = eMainCtrl;
135 Element* controlmap_el = rotor_element->FindElement("controlmap");
136 if (controlmap_el) {
137 string cm = rotor_element->FindElementValue("controlmap");
138 cm = to_upper(cm);
139 if (cm == "TAIL") {
140 ControlMap = eTailCtrl;
141 } else if (cm == "TANDEM") {
142 ControlMap = eTandemCtrl;
143 } else {
144 FGXMLLogging log(controlmap_el, LogLevel::ERROR);
145 log << "# found unknown controlmap: '" << cm << "' using main rotor config.\n";
146 }
147 }
148
149 // ExternalRPM -- is the RPM dictated ?
150 Element* extrpm_el = rotor_element->FindElement("ExternalRPM");
151 if (extrpm_el) {
152 ExternalRPM = 1;
153 SourceGearRatio = 1.0;
154 RPMdefinition = (int) rotor_element->FindElementValueAsNumber("ExternalRPM");
155 int rdef = RPMdefinition;
156 if (RPMdefinition>=0) {
157 // avoid ourself and (still) unknown engines.
158 if (!exec->GetPropulsion()->GetEngine(RPMdefinition) || RPMdefinition==num) {
159 RPMdefinition = -1;
160 } else {
161 FGThruster *tr = exec->GetPropulsion()->GetEngine(RPMdefinition)->GetThruster();
162 SourceGearRatio = tr->GetGearRatio();
163 //FGLogging log(LogLevel::INFO);
164 //log << "# got sources' GearRatio: " << SourceGearRatio << "\n";
165 }
166 }
167 if (RPMdefinition != rdef) {
168 FGXMLLogging log(extrpm_el, LogLevel::ERROR);
169 log << "# discarded given RPM source (" << rdef
170 << ") and switched to external control (-1).\n";
171 }
172 }
173
174 // process rotor parameters
175 engine_power_est = Configure(rotor_element);
176
177 // setup transmission if needed
178 if (!ExternalRPM) {
179
180 Transmission = new FGTransmission(exec, num, dt);
181
182 Transmission->SetThrusterMoment(PolarMoment);
183
184 // The MOI sensed behind the gear ( MOI_engine*sqr(GearRatio) ).
185 GearMoment = ConfigValueConv(rotor_element, "gearmoment", 0.1*PolarMoment, "SLUG*FT2");
186 GearMoment = Constrain(1e-6, GearMoment, 1e9);
187 Transmission->SetEngineMoment(GearMoment);
188
189 Transmission->SetMaxBrakePower(MaxBrakePower);
190
191 GearLoss = ConfigValueConv(rotor_element, "gearloss", 0.0025 * engine_power_est, "HP");
192 GearLoss = Constrain(0.0, GearLoss, 1e9);
193 GearLoss *= hptoftlbssec;
194 Transmission->SetEngineFriction(GearLoss);
195
196 }
197
198 // shaft representation - a rather simple transform,
199 // but using a matrix is safer.
200 TboToHsr = { 0.0, 0.0, 1.0,
201 0.0, 1.0, 0.0,
202 -1.0, 0.0, 0.0 };
203 HsrToTbo = TboToHsr.Transposed();
204
205 // smooth out jumps in hagl reported, otherwise the ground effect
206 // calculation would cause jumps too. 1Hz seems sufficient.
207 damp_hagl = Filter(1.0, dt);
208
209 // enable import-export
210 bindmodel(exec->GetPropertyManager().get());
211
212 Debug(0);
213
214} // Constructor

278 }
279 }
280
281 return val;
282}

293// 1. read configuration and try to fill holes, ymmv
294// 2. calculate derived parameters
295double FGRotor::Configure(Element* rotor_element)
296{
297
298 double estimate, engine_power_est=0.0;
299 const bool yell = true;
300 const bool silent = false;
301
302
303 Radius = 0.5 * ConfigValueConv(rotor_element, "diameter", 42.0, "FT", yell);
304 Radius = Constrain(1e-3, Radius, 1e9);
305
306 BladeNum = (int) ConfigValue(rotor_element, "numblades", 3 , yell);
307
308 GearRatio = ConfigValue(rotor_element, "gearratio", 1.0, yell);
309 GearRatio = Constrain(1e-9, GearRatio, 1e9);
310
311 // make sure that v_tip (omega*r) is below 0.7mach ~ 750ft/s
312 estimate = (750.0/Radius)/(2.0*M_PI) * 60.0; // 7160/Radius
313 NominalRPM = ConfigValue(rotor_element, "nominalrpm", estimate, yell);
314 NominalRPM = Constrain(2.0, NominalRPM, 1e9);
315
316 MinimalRPM = ConfigValue(rotor_element, "minrpm", 1.0);
317 MinimalRPM = Constrain(1.0, MinimalRPM, NominalRPM - 1.0);
318
319 MaximalRPM = ConfigValue(rotor_element, "maxrpm", 2.0*NominalRPM);
320 MaximalRPM = Constrain(NominalRPM, MaximalRPM, 1e9);
321
322 estimate = Constrain(0.07, 2.0/Radius , 0.14); // guess solidity
323 estimate = estimate * M_PI*Radius/BladeNum;
324 BladeChord = ConfigValueConv(rotor_element, "chord", estimate, "FT", yell);
325
326 LiftCurveSlope = ConfigValue(rotor_element, "liftcurveslope", 6.0); // "1/RAD"
327 BladeTwist = ConfigValueConv(rotor_element, "twist", -0.17, "RAD");
328
329 HingeOffset = ConfigValueConv(rotor_element, "hingeoffset", 0.05 * Radius, "FT" );
330
331 estimate = sqr(BladeChord) * sqr(Radius - HingeOffset) * 0.57;
332 BladeFlappingMoment = ConfigValueConv(rotor_element, "flappingmoment", estimate, "SLUG*FT2");
333 BladeFlappingMoment = Constrain(1e-9, BladeFlappingMoment, 1e9);
334
335 // guess mass from moment of a thin stick, and multiply by the blades cg distance
336 estimate = ( 3.0 * BladeFlappingMoment / sqr(Radius) ) * (0.45 * Radius) ;
337 BladeMassMoment = ConfigValue(rotor_element, "massmoment", estimate); // unit is slug-ft
338 BladeMassMoment = Constrain(1e-9, BladeMassMoment, 1e9);
339
340 estimate = 1.1 * BladeFlappingMoment * BladeNum;
341 PolarMoment = ConfigValueConv(rotor_element, "polarmoment", estimate, "SLUG*FT2");
342 PolarMoment = Constrain(1e-9, PolarMoment, 1e9);
343
344 // "inflowlag" is treated further down.
345
346 TipLossB = ConfigValue(rotor_element, "tiplossfactor", 1.0, silent);
347
348 // estimate engine power (bit of a pity, cause our caller already knows)
349 engine_power_est = 0.5 * BladeNum*BladeChord*Radius*Radius;
350
351 estimate = engine_power_est / 30.0;
352 MaxBrakePower = ConfigValueConv(rotor_element, "maxbrakepower", estimate, "HP");
353 MaxBrakePower *= hptoftlbssec;
354
355 GroundEffectExp = ConfigValue(rotor_element, "groundeffectexp", 0.0);
356 GroundEffectShift = ConfigValueConv(rotor_element, "groundeffectshift", 0.0, "FT");
357
358 // precalc often used powers
359 R[0]=1.0; R[1]=Radius; R[2]=R[1]*R[1]; R[3]=R[2]*R[1]; R[4]=R[3]*R[1];
360 B[0]=1.0; B[1]=TipLossB; B[2]=B[1]*B[1]; B[3]=B[2]*B[1]; B[4]=B[3]*B[1];
361
362 // derived parameters
363 LockNumberByRho = LiftCurveSlope * BladeChord * R[4] / BladeFlappingMoment;
364 Solidity = BladeNum * BladeChord / (M_PI * Radius);
365
366 // estimate inflow lag, see /GE49/ eqn(1)
367 double omega_tmp = (NominalRPM/60.0)*2.0*M_PI;
368 estimate = 16.0/(LockNumberByRho*rho * omega_tmp ); // 16/(gamma*Omega)
369 // printf("# Est. InflowLag: %f\n", estimate);
370 InflowLag = ConfigValue(rotor_element, "inflowlag", estimate, yell);
371 InflowLag = Constrain(1e-6, InflowLag, 2.0);
372
373 return engine_power_est;
374} // Configure

378// calculate control-axes components of total airspeed at the hub.
379// sets rotor orientation angle (beta) as side effect. /SH79/ eqn(19-22)
380
381FGColumnVector3 FGRotor::hub_vel_body2ca( const FGColumnVector3 &uvw,
382 const FGColumnVector3 &pqr,
383 double a_ic, double b_ic)
384{
385 FGColumnVector3 v_r, v_shaft, v_w;
386 FGColumnVector3 pos;
387
388 pos = fdmex->GetMassBalance()->StructuralToBody(GetActingLocation());
389
390 v_r = uvw + pqr*pos;
391 v_shaft = TboToHsr * InvTransform * v_r;
392
393 beta_orient = atan2(v_shaft(eV),v_shaft(eU));
394
395 v_w(eU) = v_shaft(eU)*cos(beta_orient) + v_shaft(eV)*sin(beta_orient);
396 v_w(eV) = 0.0;
397 v_w(eW) = v_shaft(eW) - b_ic*v_shaft(eU) - a_ic*v_shaft(eV);
398
399 return v_w;
400}

404// express fuselage angular velocity in control axes /SH79/ eqn(30,31)
405
406FGColumnVector3 FGRotor::fus_angvel_body2ca( const FGColumnVector3 &pqr)
407{
408 FGColumnVector3 av_s_fus, av_w_fus;
409
410 // for comparison:
411 // av_s_fus = BodyToShaft * pqr; /SH79/
412 // BodyToShaft = TboToHsr * InvTransform
413 av_s_fus = TboToHsr * InvTransform * pqr;
414
415 av_w_fus(eP)= av_s_fus(eP)*cos(beta_orient) + av_s_fus(eQ)*sin(beta_orient);
416 av_w_fus(eQ)= - av_s_fus(eP)*sin(beta_orient) + av_s_fus(eQ)*cos(beta_orient);
417 av_w_fus(eR)= av_s_fus(eR);
418
419 return av_w_fus;
420}

423//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
424
425// The calculation is a bit tricky because thrust depends on induced velocity,
426// and vice versa.
427//
428// The flow_scale parameter (ranging from 0.5-1.0) is used to approximate a
429// reduction of inflow if the helicopter is close to the ground, yielding to
430// higher thrust, see /TA77/ eqn(10a).
431
432void FGRotor::calc_flow_and_thrust( double theta_0, double Uw, double Ww,
433 double flow_scale)
434{
435
436 double ct_over_sigma = 0.0;
437 double c0, ct_l, ct_t0, ct_t1;
438 double mu2;
439
440 mu = Uw/(Omega*Radius); // /SH79/ eqn(24)
441 if (mu > 0.7) mu = 0.7;
442 mu2 = sqr(mu);
443
444 ct_t0 = (1.0/3.0*B[3] + 1.0/2.0 * TipLossB*mu2 - 4.0/(9.0*M_PI) * mu*mu2 ) * theta_0;
445 ct_t1 = (1.0/4.0*B[4] + 1.0/4.0 * B[2]*mu2) * BladeTwist;
446
447 ct_l = (1.0/2.0*B[2] + 1.0/4.0 * mu2) * lambda; // first time
448
449 c0 = (LiftCurveSlope/2.0)*(ct_l + ct_t0 + ct_t1) * Solidity;
450 c0 = c0 / ( 2.0 * sqrt( sqr(mu) + sqr(lambda) ) + 1e-15);
451
452 // replacement for /SH79/ eqn(26).
453 // ref: dnu/dt = 1/tau ( Ct / (2*sqrt(mu^2+lambda^2)) - nu )
454 // taking mu and lambda constant, this integrates to
455
456 nu = flow_scale * ((nu - c0) * exp(-dt/InflowLag) + c0);
457
458 // now from nu to lambda, C_T, and Thrust
459
460 lambda = Ww/(Omega*Radius) - nu; // /SH79/ eqn(25)
461
462 ct_l = (1.0/2.0*B[2] + 1.0/4.0 * mu2) * lambda;
463
464 ct_over_sigma = (LiftCurveSlope/2.0)*(ct_l + ct_t0 + ct_t1); // /SH79/ eqn(27)
465
466 Thrust = BladeNum*BladeChord*Radius*rho*sqr(Omega*Radius) * ct_over_sigma;
467
468 C_T = ct_over_sigma * Solidity;
469 v_induced = nu * (Omega*Radius);
470
471}

474//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
475
476// Two blade teetering rotors are often 'preconed' to a fixed angle, but the
477// calculated value is pretty close to the real one. /SH79/ eqn(29)
478
479void FGRotor::calc_coning_angle(double theta_0)
480{
481 double lock_gamma = LockNumberByRho * rho;
482
483 double a0_l = (1.0/6.0 + 0.04 * mu*mu*mu) * lambda;
484 double a0_t0 = (1.0/8.0 + 1.0/8.0 * mu*mu) * theta_0;
485 double a0_t1 = (1.0/10.0 + 1.0/12.0 * mu*mu) * BladeTwist;
486 a0 = lock_gamma * ( a0_l + a0_t0 + a0_t1);
487 return;
488}

490//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
491
492// Flapping angles relative to control axes /SH79/ eqn(32)
493
494void FGRotor::calc_flapping_angles(double theta_0, const FGColumnVector3 &pqr_fus_w)
495{
496 double lock_gamma = LockNumberByRho * rho;
497
498
499 double mu2_2 = sqr(mu)/2.0;
500 double t075 = theta_0 + 0.75 * BladeTwist; // common approximation for rectangular blades
501
502 a_1 = 1.0/(1.0 - mu2_2) * (
503 (2.0*lambda + (8.0/3.0)*t075)*mu
504 + pqr_fus_w(eP)/Omega
505 - 16.0 * pqr_fus_w(eQ)/(lock_gamma*Omega)
506 );
507
508 b_1 = 1.0/(1.0 + mu2_2) * (
509 (4.0/3.0)*mu*a0
510 - pqr_fus_w(eQ)/Omega
511 - 16.0 * pqr_fus_w(eP)/(lock_gamma*Omega)
512 );
513
514 // used in force calc
515 a_dw = 1.0/(1.0 - mu2_2) * (
516 (2.0*lambda + (8.0/3.0)*t075)*mu
517 - 24.0 * pqr_fus_w(eQ)/(lock_gamma*Omega)
518 * ( 1.0 - ( 0.29 * t075 / (C_T/Solidity) ) )
519 );
520
521 return;
522}

524//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
525
526// /SH79/ eqn(38,39)
527
528void FGRotor::calc_drag_and_side_forces(double theta_0)
529{
530 double cy_over_sigma;
531 double t075 = theta_0 + 0.75 * BladeTwist;
532
533 H_drag = Thrust * a_dw;
534
535 cy_over_sigma = (
536 0.75*b_1*lambda - 1.5*a0*mu*lambda + 0.25*a_1*b_1*mu
537 - a0*a_1*sqr(mu) + (1.0/6.0)*a0*a_1
538 - (0.75*mu*a0 - (1.0/3.0)*b_1 - 0.5*sqr(mu)*b_1)*t075
539 );
540 cy_over_sigma *= LiftCurveSlope/2.0;
541
542 J_side = BladeNum * BladeChord * Radius * rho * sqr(Omega*Radius) * cy_over_sigma;
543
544 return;
545}

547//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
548
549// Simplified version of /SH79/ eqn(36). Uses an estimate for blade drag
550// (a new config parameter to come...).
551// From "Bramwell's Helicopter Dynamics", second edition, eqn(3.43) and (3.44)
552
553void FGRotor::calc_torque(double theta_0)
554{
555 // estimate blade drag
556 double delta_dr = 0.009 + 0.3*sqr(6.0*C_T/(LiftCurveSlope*Solidity));
557
558 Torque = rho * BladeNum * BladeChord * delta_dr * sqr(Omega*Radius) * R[2] *
559 (1.0+4.5*sqr(mu))/8.0
560 - (Thrust*lambda + H_drag*mu)*Radius;
561
562 return;
563}

565//%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
566
567// Get the downwash angles with respect to the shaft axis.
568// Given a 'regular' main rotor, the angles are zero when the downwash points
569// down, positive theta values mean that the downwash turns towards the nose,
570// and positive phi values mean it turns to the left side. (Note: only airspeed
571// is transformed, the rotational speed contribution is ignored.)
572
573void FGRotor::calc_downwash_angles()
574{
575 FGColumnVector3 v_shaft;
576 v_shaft = TboToHsr * InvTransform * in.AeroUVW;
577
578 theta_downwash = atan2( -v_shaft(eU), v_induced - v_shaft(eW)) + a1s;
579 phi_downwash = atan2( v_shaft(eV), v_induced - v_shaft(eW)) + b1s;
580
581 return;
582}

585
586// transform rotor forces from control axes to shaft axes, and express
587// in body axes /SH79/ eqn(40,41)
588
589FGColumnVector3 FGRotor::body_forces(double a_ic, double b_ic)
590{
591 FGColumnVector3 F_s(
592 - H_drag*cos(beta_orient) - J_side*sin(beta_orient) + Thrust*b_ic,
593 - H_drag*sin(beta_orient) + J_side*cos(beta_orient) + Thrust*a_ic,
594 - Thrust);
595
596 return HsrToTbo * F_s;
597}

601// calculates the additional moments due to hinge offset and handles
602// torque and sense
603
604FGColumnVector3 FGRotor::body_moments(double a_ic, double b_ic)
605{
606 FGColumnVector3 M_s, M_hub, M_h;
607 double mf;
608
609 // cyclic flapping relative to shaft axes /SH79/ eqn(43)
610 a1s = a_1*cos(beta_orient) + b_1*sin(beta_orient) - b_ic;
611 b1s = b_1*cos(beta_orient) - a_1*sin(beta_orient) + a_ic;
612
613 mf = 0.5 * HingeOffset * BladeNum * Omega*Omega * BladeMassMoment;
614
615 M_s(eL) = mf*b1s;
616 M_s(eM) = mf*a1s;
617 M_s(eN) = Torque * Sense ;
618
619 return HsrToTbo * M_s;
620}

623
624void FGRotor::CalcRotorState(void)
625{
626 double A_IC; // lateral (roll) control in radians
627 double B_IC; // longitudinal (pitch) control in radians
628 double theta_col; // rotor collective pitch in radians
629
630 FGColumnVector3 vHub_ca, avFus_ca;
631
632 double filtered_hagl = 0.0;
633 double ge_factor = 1.0;
634
635 // fetch needed values from environment
636 rho = in.Density; // slugs/ft^3.
637 double h_agl_ft = in.H_agl;
638
639 // update InvTransform, the rotor orientation could have been altered
640 InvTransform = Transform().Transposed();
641
642 // handle RPM requirements, calc omega.
643 if (ExternalRPM && ExtRPMsource) {
644 RPM = ExtRPMsource->getDoubleValue() * ( SourceGearRatio / GearRatio );
645 }
646
647 // MinimalRPM is always >= 1. MaximalRPM is always >= NominalRPM
648 RPM = Constrain(MinimalRPM, RPM, MaximalRPM);
649
650 Omega = (RPM/60.0)*2.0*M_PI;
651
652 // set control inputs
653 A_IC = LateralCtrl;
654 B_IC = LongitudinalCtrl;
655 theta_col = CollectiveCtrl;
656
657 // optional ground effect, a ge_factor of 1.0 gives no effect
658 // and 0.5 yields to maximal influence.
659 if (GroundEffectExp > 1e-5) {
660 if (h_agl_ft<0.0) h_agl_ft = 0.0; // clamp
661 filtered_hagl = damp_hagl.execute(h_agl_ft) + GroundEffectShift;
662 // actual/nominal factor avoids absurd scales at startup
663 ge_factor -= GroundEffectScaleNorm *
664 ( exp(-filtered_hagl*GroundEffectExp) * (RPM / NominalRPM) );
665 ge_factor = Constrain(0.5, ge_factor, 1.0);
666 }
667
668 // all set, start calculations ...
669
670 vHub_ca = hub_vel_body2ca(in.AeroUVW, in.AeroPQR, A_IC, B_IC);
671
672 avFus_ca = fus_angvel_body2ca(in.AeroPQR);
673
674 calc_flow_and_thrust(theta_col, vHub_ca(eU), vHub_ca(eW), ge_factor);
675
676 calc_coning_angle(theta_col);
677
678 calc_flapping_angles(theta_col, avFus_ca);
679
680 calc_drag_and_side_forces(theta_col);
681
682 calc_torque(theta_col);
683
684 calc_downwash_angles();
685
686 // ... and assign to inherited vFn and vMn members
687 // (for processing see FGForce::GetBodyForces).
688 vFn = body_forces(A_IC, B_IC);
689 vMn = Transform() * body_moments(A_IC, B_IC);
690
691}

694
695double FGRotor::Calculate(double EnginePower)
696{
697
698 CalcRotorState();
699
700 if (! ExternalRPM) {
701 // the RPM values are handled inside Transmission
702 Transmission->Calculate(EnginePower, Torque, in.TotalDeltaT);
703
704 EngineRPM = Transmission->GetEngineRPM() * GearRatio;
705 RPM = Transmission->GetThrusterRPM();
706 } else {
707 EngineRPM = RPM * GearRatio;
708 }
709
710 RPM = Constrain(MinimalRPM, RPM, MaximalRPM); // trim again
711
712 return Thrust;
713}

840// The bitmasked value choices are as follows:
841// unset: In this case (the default) JSBSim would only print
842// out the normally expected messages, essentially echoing
843// the config files as they are read. If the environment
844// variable is not set, debug_lvl is set to 1 internally
845// 0: This requests JSBSim not to output any messages
846// whatsoever.
847// 1: This value explicity requests the normal JSBSim
848// startup messages
849// 2: This value asks for a message to be printed out when
850// a class is instantiated
851// 4: When this value is set, a message is displayed when a
852// FGModel object executes its Run() method
853// 8: When this value is set, various runtime state variables
854// are printed out periodically
855// 16: When set various parameters are sanity checked and
856// a message is printed out when they go out of bounds
